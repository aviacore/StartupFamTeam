**routes/agents.js:**

```javascript
const express = require('express');
const router = express.Router();
const fs = require('fs');
const path = require('path');
const { validateResponse } = require('../middleware/validator');

const DATA_FILE = path.join(__dirname, '../data/agent_responses.json');

// Загрузка или создание файла данных
function loadData() {
  if (!fs.existsSync(DATA_FILE)) {
    fs.writeFileSync(DATA_FILE, JSON.stringify([], null, 2));
  }
  return JSON.parse(fs.readFileSync(DATA_FILE, 'utf8'));
}

function saveData(data) {
  fs.writeFileSync(DATA_FILE, JSON.stringify(data, null, 2));
}

// POST: Принять ответ от агента
router.post('/response', (req, res) => {
  const validation = validateResponse(req.body);
  if (!validation.valid) {
    return res.status(400).json({ error: validation.errors.join(', ') });
  }

  const data = loadData();
  data.push({
    ...req.body,
    timestamp: new Date().toISOString()
  });
  saveData(data);

  res.status(201).json({ message: 'Ответ агента сохранён', agent: req.body.agent });
});

// GET: Получить все ответы
router.get('/responses', (req, res) => {
  const data = loadData();
  res.json(data);
});

// GET: Сгенерировать Plan_06.md
router.get('/plan06', (req, res) => {
  try {
    const data = loadData();
    if (data.length === 0) {
      return res.status(404).json({ error: 'Нет данных агентов' });
    }

    // Генерация Markdown
    let tableRows = data.map(r => 
      `| ${r.agent} | ${r.russian_title} | ${r.synonym || '-'} |`
    ).join('\n');

    const markdown = `# Plan_06 — Названия должностей/скиллов агентов (русский язык)

**Составитель:** Developer (автоматическая генерация через API)
**Дата:** ${new Date().toISOString().split('T')[0]}

## Сводка

| Агент (англ. навык) | Название на русском языке (основное) | Вариант (синоним) |
|----------------------|---------------------------------------|-------------------|
${tableRows}

## Примечание
Файл сгенерирован автоматически API сервером на основе данных из /api/agents/responses.
`;

    // Сохраняем и возвращаем
    const outputPath = path.join(__dirname, '../../Plan_06.md');
    fs.writeFileSync(outputPath, markdown, 'utf8');
    
    res.json({ 
      message: 'Plan_06.md сгенерирован', 
      path: outputPath,
      markdown: markdown 
    });
  } catch (err) {
    res.status(500).json({ error: `Ошибка генерации: ${err.message}` });
  }
});

module.exports = router;
```

**middleware/validator.js:**

```javascript
const schema = {
  type: 'object',
  required: ['agent', 'russian_title'],
  properties: {
    agent: { type: 'string', enum: ['Ideation', 'Strategy', 'Business Model', 'Pitch'] },
    english_skill: { type: 'string' },
    russian_title: { type: 'string', minLength: 1 },
    synonym: { type: 'string' }
  }
};

function validateResponse(data) {
  const errors = [];
  
  if (!data.agent || !schema.properties.agent.enum.includes(data.agent)) {
    errors.push('Неверное имя агента. Допустимые: Ideation, Strategy, Business Model, Pitch');
  }
  
  if (!data.russian_title || data.russian_title.trim() === '') {
    errors.push('Поле russian_title обязательно и не может быть пустым');
  }

  return { valid: errors.length === 0, errors };
}

module.exports = { validateResponse };
```

### 2. Интерактивный CLI-клиент для CEO

Вместо ручного копирования данных, я предлагаю CLI-инструмент, который CEO может запустить в терминале и сразу получить все ответы и сгенерировать Plan_06.

**project/scripts/ceo_collect.sh:**

```bash
#!/bin/bash
# CLI для CEO: сбор ответов от агентов в интерактивном режиме

echo "=== Сбор названий агентов (русский язык) ==="
echo ""

# Массив агентов и их ID
declare -A agents
agents[1]="Ideation"
agents[2]="Strategy"
agents[3]="Business Model"
agents[4]="Pitch"

responses_file="project/data/agent_responses.json"

# Инициализация файла если пуст
if [ ! -f "$responses_file" ]; then
  echo "[]" > "$responses_file"
fi

for i in {1..4}; do
  agent=${agents[$i]}
  echo "Агент: $agent"
  echo -n "Введите название на русском (основное): "
  read russian_title
  
  echo -n "Введите синоним (опционально, Enter для пропуска): "
  read synonym
  
  # Формируем JSON и добавляем в файл
  timestamp=$(date -u +"%Y-%m-%dT%H:%M:%SZ")
  
  if [ -z "$synonym" ]; then
    entry=$(cat <<EOF
{
  "agent": "$agent",
  "english_skill": "$agent",
  "russian_title": "$russian_title",
  "timestamp": "$timestamp"
}
EOF
    )
  else
    entry=$(cat <<EOF
{
  "agent": "$agent",
  "english_skill": "$agent",
  "russian_title": "$russian_title",
  "synonym": "$synonym",
  "timestamp": "$timestamp"
}
EOF
    )
  fi
  
  # Добавляем к файлу (простая реализация - перезапись)
  current=$(cat "$responses_file")
  if [ "$current" = "[]" ]; then
    new="[$entry]"
  else
    new=$(echo "$current" | sed "s/\]$/, $entry\]/")
  fi
  echo "$new" > "$responses_file"
  
  echo "✓ Ответ агента $agent сохранён"
  echo ""
done

# Генерация Plan_06
echo "Генерирую Plan_06.md..."
python3 project/scripts/generate_plan_06.py

echo ""
echo "=== Готово! Файл Plan_06.md создан ==="
```

```html
<!DOCTYPE html>
<html lang="ru">

    .success { color: green; }

</head>
<body>
  <h1>Сбор названий агентов (русский язык)</h1>
  
  <div id="status"></div>
  
  <h2>Текущие ответы</h2>
  <table id="responses-table">
    <thead>
      <tr>
        <th>Агент</th>
        <th>Название (рус.)</th>
        <th>Синоним</th>
        <th>Дата</th>
      </tr>
    </thead>
    <tbody id="responses-body">
      <tr><td colspan="4">Загрузка...</td></tr>
    </tbody>
  </table>

  <button class="btn" onclick="generatePlan()">🔄 Сгенерировать Plan_06.md</button>
  <button class="btn" onclick="clearAll()">🗑️ Очистить все данные</button>

  <script>
    async function loadResponses() {
      try {
        const res = await fetch('/api/agents/responses');
        const data = await res.json();
        
        const tbody = document.getElementById('responses-body');
        tbody.innerHTML = '';
        
        if (data.length === 0) {
          tbody.innerHTML = '<tr><td colspan="4">Нет данных. Используйте CLI или API для добавления.</td></tr>';
          return;
        }
        
        data.forEach(item => {
          const row = document.createElement('tr');
          row.innerHTML = `
            <td>${item.agent}</td>
            <td>${item.russian_title}</td>
            <td>${item.synonym || '-'}</td>
            <td>${item.timestamp ? item.timestamp.slice(0, 10) : '-'}</td>
          `;
          tbody.appendChild(row);
        });
      } catch (err) {
        console.error('Ошибка загрузки:', err);
        document.getElementById('status').innerHTML = '<p class="error">❌ Ошибка загрузки данных</p>';
      }
    }

    async function generatePlan() {
      document.getElementById('status').innerHTML = '<p>Генерация...</p>';
      try {
        const res = await fetch('/api/agents/plan06');
        const result = await res.json();
        if (res.ok) {
          document.getElementById('status').innerHTML = `<p class="success">✅ Plan_06.md сгенерирован: ${result.path}</p>`;
        } else {
          document.getElementById('status').innerHTML = `<p class="error">❌ ${result.error}</p>`;
        }
      } catch (err) {
        document.getElementById('status').innerHTML = `<p class="error">❌ ${err.message}</p>`;
      }
    }

    async function clearAll() {
      if (!confirm('Удалить все ответы агентов?')) return;
      try {
        await fetch('/api/agents/responses', { method: 'DELETE' });
        document.getElementById('status').innerHTML = '<p class="success">✅ Данные очищены</p>';
        loadResponses();
      } catch (err) {
        document.getElementById('status').innerHTML = `<p class="error">❌ ${err.message}</p>`;
      }
    }

    loadResponses();
  </script>
</body>
</html>
```
