## 1. Критика существующего прототипа кода (от CTO)

Предложенный CTO фрагмент кода — это **слишком упрощённая версия**. Он создаёт новый WebSocket-сервер при каждом `POST /roll-call`, что приводит к:
- **Утечке портов:** Каждый вызов открывает новый сервер на порту 8081, старый не закрывается.
- **Отсутствию идентификации:** Нет механизма, который бы сообщил агентам, на какой порт и сервер подключаться.
- **Нет таймаутов:** Нет проверки, что агент не ответил за 10 секунд (как просил PM).

**Моё решение:** Сделать один статический WebSocket-сервер, который живёт всё время. Агенты подключаются к нему один раз при старте. Координатор (CEO) отправляет команду через REST API, а сервер транслирует её всем подключенным агентам.

## 2. Детальная спецификация API и WebSocket протокола

Чтобы все компоненты (фронт, бэк, агенты) работали согласованно, нужен чёткий протокол:

### REST API (для CEO/фронта)
| Endpoint | Метод | Описание |
|----------|-------|----------|
| `/api/roll-call` | POST | Запускает новую перекличку. Возвращает `{ "rollCallId": 1 }` |
| `/api/status` | GET | Возвращает текущий статус последней переклички |
| `/api/history` | GET | Возвращает историю всех перекличек |

### WebSocket (для агентов)
**Сообщения от сервера к агенту:**
```json
{
  "type": "ping",
  "rollCallId": 1,
  "timestamp": "2024-01-15T10:00:00Z"
}
```

**Сообщения от агента к серверу:**
```json
{
  "type": "pong",
  "rollCallId": 1,
  "role": "Идеатор",
  "status": "готов",
  "message": "Жду новой задачи",
  "timestamp": "2024-01-15T10:00:01Z"
}
```

## 4. Реализация MVP: готовый код трёх ключевых файлов

Я представляю прототипы кода, которые сразу можно запустить:

### `server.js` (бэкенд)
```javascript
const express = require('express');
const http = require('http');
const WebSocket = require('ws');
const { initDB, saveRollCall, saveResponse, getAgents } = require('./database');
const { startRollCall } = require('./roll-call-manager');

const app = express();
const server = http.createServer(app);
const wss = new WebSocket.Server({ server });

app.use(express.json());
app.use(express.static('public'));

// Хранилище подключенных агентов: { ws -> { role, connectedAt } }
const connectedAgents = new Map();

wss.on('connection', (ws) => {
  console.log('Новое WebSocket подключение');

  ws.on('message', (message) => {
    const data = JSON.parse(message);
    
    if (data.type === 'register') {
      // Агент регистрируется
      connectedAgents.set(ws, { role: data.role, connectedAt: new Date() });
      ws.send(JSON.stringify({ type: 'registered', role: data.role }));
    }
  });

  ws.on('close', () => {
    connectedAgents.delete(ws);
  });
});

// API для CEO
app.post('/api/roll-call', async (req, res) => {
  const db = initDB();
  const rollCallId = saveRollCall(db);
  
  // Запускаем асинхронно (ответ сразу)
  startRollCall(wss, connectedAgents, rollCallId, db);
  
  res.json({ rollCallId });
});

app.get('/api/status', (req, res) => {
  const db = initDB();
  const lastRollCall = db.prepare('SELECT * FROM roll_calls ORDER BY id DESC LIMIT 1').get();
  res.json(lastRollCall || { status: 'none' });
});

app.get('/api/history', (req, res) => {
  const db = initDB();
  const rollCalls = db.prepare('SELECT * FROM roll_calls ORDER BY id DESC LIMIT 10').all();
  res.json(rollCalls);
});

const PORT = process.env.PORT || 3000;
server.listen(PORT, () => {
  console.log(`Сервер запущен на порту ${PORT}`);
  initDB();
});
```

### `roll-call-manager.js` (логика переклички)
```javascript
function startRollCall(wss, connectedAgents, rollCallId, db) {
  const TIMEOUT_MS = 10000; // 10 секунд
  
  // 1. Устанавливаем статус переклички
  db.prepare('UPDATE roll_calls SET status = ? WHERE id = ?').run('open', rollCallId);
  
  // 2. Отправляем ping всем подключенным агентам
  const pingMessage = JSON.stringify({
    type: 'ping',
    rollCallId,
    timestamp: new Date().toISOString()
  });
  
  connectedAgents.forEach((agentInfo, ws) => {
    if (ws.readyState === WebSocket.OPEN) {
      ws.send(pingMessage);
    }
  });
  
  // 3. Устанавливаем таймер на закрытие переклички
  setTimeout(() => {
    // Проверяем, кто ответил
    const responses = db.prepare('SELECT * FROM roll_call_responses WHERE roll_call_id = ?').all(rollCallId);
    const answeredRoles = new Set(responses.map(r => r.agent_role));
    
    // Отмечаем неответивших как offline
    connectedAgents.forEach((agentInfo) => {
      if (!answeredRoles.has(agentInfo.role)) {
        db.prepare('INSERT INTO roll_call_responses (roll_call_id, agent_role, parsed_status) VALUES (?, ?, ?)').run(rollCallId, agentInfo.role, 'offline');
      }
    });
    
    // Закрываем перекличку
    db.prepare('UPDATE roll_calls SET status = ? WHERE id = ?').run('closed', rollCallId);
    console.log(`Перекличка ${rollCallId} завершена`);
  }, TIMEOUT_MS);
}

// Обработка ответа агента
function processAgentResponse(ws, data, db) {
  const { rollCallId, role, status, message } = data;
  
  // Сохраняем в БД
  db.prepare(
    'INSERT INTO roll_call_responses (roll_call_id, agent_role, response_text, parsed_status) VALUES (?, ?, ?, ?)'
  ).run(rollCallId, role, message || '', status);
  
  // Отправляем подтверждение агенту
  ws.send(JSON.stringify({ type: 'ack', rollCallId }));
}

module.exports = { startRollCall, processAgentResponse };
```

### `agents/ideation.js` (пример агента)
```javascript
const WebSocket = require('ws');

const SERVER_URL = 'ws://localhost:3000';
const ROLE = 'Идеатор';

const ws = new WebSocket(SERVER_URL);

const responses = [
  'У меня есть несколько свежих идей на основе твоего последнего запроса',
  'Жду новой задачи для генерации',
  'Анализирую текущие тренды рынка'
];

ws.on('open', () => {
  console.log(`Агент "${ROLE}" подключился к серверу`);
  
  // Регистрируемся
  ws.send(JSON.stringify({
    type: 'register',
    role: ROLE
  }));
});

ws.on('message', (message) => {
  const data = JSON.parse(message);
  
  if (data.type === 'ping') {
    console.log(`Получен ping для переклички #${data.rollCallId}`);
    
    // Имитируем задержку (100-500 мс) как реальный агент
    const delay = Math.floor(Math.random() * 400) + 100;
    setTimeout(() => {
      ws.send(JSON.stringify({
        type: 'pong',
        rollCallId: data.rollCallId,
        role: ROLE,
        status: 'готов',
        message: responses[Math.floor(Math.random() * responses.length)]
      }));
    }, delay);
  }
  
  if (data.type === 'ack') {
    console.log(`Ответ принят сервером для переклички #${data.rollCallId}`);
  }
});
```

### `public/app.js`
```javascript
const API_URL = window.location.origin || 'http://localhost:3000';

async function startRollCall() {
  const btn = document.getElementById('roll-call-btn');
  const statusEl = document.getElementById('status-text');
  const resultsEl = document.getElementById('results');
  const resultsList = document.getElementById('results-list');
  
  btn.disabled = true;
  btn.textContent = '⏳ Проводим перекличку...';
  statusEl.textContent = 'Запускаю опрос агентов...';
  resultsEl.style.display = 'none';
  
  try {
    // 1. Запускаем перекличку
    const response = await fetch(`${API_URL}/api/roll-call`, { method: 'POST' });
    const { rollCallId } = await response.json();
    
    statusEl.textContent = `Перекличка #${rollCallId} запущена. Жду ответы агентов...`;
    
    // 2. Ждём 12 секунд (10 секунд таймаут + запас)
    await new Promise(resolve => setTimeout(resolve, 12000));
    
    // 3. Получаем статус
    const statusResponse = await fetch(`${API_URL}/api/status`);
    const status = await statusResponse.json();
    
    // 4. Получаем результаты из GET запроса с параметром
    const historyResponse = await fetch(`${API_URL}/api/history`);
    const history = await historyResponse.json();
    
    // Очищаем и заполняем результаты
    resultsList.innerHTML = '';
    resultsEl.style.display = 'block';
    
    // Здесь нужно сделать отдельный endpoint для получения ответов конкретной переклички
    // Пока упрощаем
    statusEl.textContent = `Перекличка #${rollCallId} завершена. Статус: ${status.status}`;
    
  } catch (error) {
    console.error('Ошибка:', error);
    statusEl.textContent = '❌ Ошибка при проведении переклички';
  } finally {
    btn.disabled = false;
    btn.textContent = '🚀 Провести перекличку';
  }
}
```
