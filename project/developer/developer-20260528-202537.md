# Developer — Output

Хорошо, давай создадим веб-приложение, которое будет представлять список популярных провайдеров/моделей с возможностью добавления адресов сайтов, API-ключей и остатка денег. Приложение будет выполняться на Node.js с использованием SQLite для хранения данных и с фронтендом на HTML, CSS и JavaScript.

   (Содержит архитектуру приложения)

   (Содержит описание задач)

### 3. Файл `script.js`

```javascript
document.getElementById('providerForm').addEventListener('submit', function(event) {
    event.preventDefault();
    
    const providerName = document.getElementById('providerName').value;
    const providerUrl = document.getElementById('providerUrl').value;
    const apiKey = document.getElementById('apiKey').value;
    const balance = document.getElementById('balance').value;

    fetch('/add-provider', {
        method: 'POST',
        headers: {
            'Content-Type': 'application/json',
        },
        body: JSON.stringify({ providerName, providerUrl, apiKey, balance }),
    })
    .then(response => response.json())
    .then(data => {
        if (data.success) {
            loadProviders();
        }
    });

    document.getElementById('providerForm').reset();
});

function loadProviders() {
    fetch('/providers')
        .then(response => response.json())
        .then(data => {
            const providerList = document.getElementById('providerList');
            providerList.innerHTML = '';
            data.providers.forEach(provider => {
                const li = document.createElement('li');
                li.textContent = `${provider.name} - ${provider.url} - API ключ: ${provider.apiKey} - Остаток: ${provider.balance}`;
                providerList.appendChild(li);
            });
        });
}

loadProviders();
```

### 4. Файл `server.js`

```javascript
const express = require('express');
const bodyParser = require('body-parser');
const { initDB, addProvider, getProviders } = require('./database');

const app = express();
const PORT = process.env.PORT || 3000;

app.use(bodyParser.json());
app.use(express.static('src'));

app.post('/add-provider', (req, res) => {
    const { providerName, providerUrl, apiKey, balance } = req.body;
    addProvider(providerName, providerUrl, apiKey, balance);
    res.json({ success: true });
});

app.get('/providers', (req, res) => {
    const providers = getProviders();
    res.json({ providers });
});

initDB().then(() => {
    app.listen(PORT, () => {
        console.log(`Сервер запущен на http://localhost:${PORT}`);
    });
});
```

### 5. Файл `database.js`

```javascript
const sqlite3 = require('sqlite3').verbose();
const db = new sqlite3.Database(':memory:');

const initDB = () => {
    return new Promise((resolve, reject) => {
        db.serialize(() => {
            db.run(`CREATE TABLE providers (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                name TEXT NOT NULL,
                url TEXT NOT NULL,
                apiKey TEXT NOT NULL,
                balance REAL NOT NULL
            )`, (err) => {
                if (err) {
                    return reject(err);
                }
                resolve();
            });
        });
    });
};

const addProvider = (name, url, apiKey, balance) => {
    db.serialize(() => {
        db.run(`INSERT INTO providers (name, url, apiKey, balance) VALUES (?, ?, ?, ?)`, [name, url, apiKey, balance]);
    });
};

const getProviders = () => {
    return new Promise((resolve, reject) => {
        db.all("SELECT * FROM providers", [], (err, rows) => {
            if (err) {
                reject(err);
            }
            resolve(rows);
        });
    });
};

module.exports = { initDB, addProvider, getProviders };
```

### Запуск приложения

1. Убедитесь, что у вас установлен Node.js и npm.
2. Создайте папку `project/src` и добавьте указанные файлы в неё.
3. Установите зависимости, выполнив команду:
   ```bash
   npm init -y
   npm install express body-parser sqlite3
   ```
4. Запустите сервер:
   ```bash
   node src/server.js
   ```
5. Откройте в браузере `http://localhost:3000`.

Теперь у вас есть готовое веб-приложение, которое позволяет добавлять провайдеров и отображать их в списке.
