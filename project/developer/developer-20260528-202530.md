# Developer — Output

Для разработки веб-приложения по заданному описанию мы создадим структуру проекта, включая файлы HTML, CSS и JavaScript для взаимодействия с пользователем, а также Node.js в качестве сервера и SQLite для работы с базой данных. 

Вот пошаговое решение:

### 1. Структура проекта

Создаем следующую структуру директории:
```
project/
│
├── src/
│   ├── index.html
│   ├── styles.css
│   ├── script.js
│
├── server.js
├── database.js
└── package.json
```

### 2. Файл `package.json`

Создайте файл `package.json` в корневом каталоге проекта:

```json
{
  "name": "provider-manager",

  "main": "server.js",
  "scripts": {
    "start": "node server.js"
```

### 3. Файл `server.js`

Создайте файл `server.js`:

```javascript
const express = require('express');
const bodyParser = require('body-parser');
const { initializeDatabase, getProviders, addProvider } = require('./database');
const app = express();
const PORT = 3000;

app.use(bodyParser.json());
app.use(bodyParser.urlencoded({ extended: true }));
app.use(express.static('src'));

initializeDatabase();

app.get('/api/providers', async (req, res) => {
    const providers = await getProviders();
    res.json(providers);
});

app.post('/api/providers', async (req, res) => {
    const { name, website, apiKey, balance } = req.body;
    await addProvider(name, website, apiKey, balance);
    res.status(201).send("Провайдер добавлен");
});

app.listen(PORT, () => {
    console.log(`Сервер запущен на http://localhost:${PORT}`);
});
```

### 4. Файл `database.js`

Создайте файл `database.js` для работы с SQLite:

```javascript
const sqlite3 = require('sqlite3').verbose();
const db = new sqlite3.Database('./providers.db');

function initializeDatabase() {
    db.serialize(() => {
        db.run(`CREATE TABLE IF NOT EXISTS providers (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            name TEXT NOT NULL,
            website TEXT NOT NULL,
            apiKey TEXT NOT NULL,
            balance REAL NOT NULL
        )`);
    });
}

function getProviders() {
    return new Promise((resolve, reject) => {
        db.all("SELECT * FROM providers", [], (err, rows) => {
            if (err) {
                reject(err);
                return;
            }
            resolve(rows);
        });
    });
}

function addProvider(name, website, apiKey, balance) {
    return new Promise((resolve, reject) => {
        db.run(`INSERT INTO providers (name, website, apiKey, balance) VALUES (?, ?, ?, ?)`, [name, website, apiKey, balance], function(err) {
            if (err) {
                reject(err);
                return;
            }
            resolve(this.lastID);
        });
    });
}

module.exports = { initializeDatabase, getProviders, addProvider };
```

### 7. Файл `script.js`

Создайте файл `src/script.js`:

```javascript
document.addEventListener("DOMContentLoaded", () => {
    const form = document.getElementById('provider-form');
    const tableBody = document.getElementById('providers-table').getElementsByTagName('tbody')[0];

    // Функция для загрузки провайдеров
    const loadProviders = async () => {
        const response = await fetch('/api/providers');
        const providers = await response.json();
        
        tableBody.innerHTML = ''; // Очистить таблицу перед добавлением

        providers.forEach(provider => {
            const row = tableBody.insertRow();
            row.innerHTML = `
                <td>${provider.name}</td>
                <td>${provider.website}</td>
                <td>${provider.apiKey}</td>
                <td>${provider.balance}</td>
            `;
        });
    };

    loadProviders();

    // Обработка отправки формы
    form.addEventListener('submit', async (e) => {
        e.preventDefault();

        const name = document.getElementById('name').value;
        const website = document.getElementById('website').value;
        const apiKey = document.getElementById('apiKey').value;
        const balance = document.getElementById('balance').value;

        await fetch('/api/providers', {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json'
            },
            body: JSON.stringify({ name, website, apiKey, balance })
        });

        form.reset(); // Сбросить форму
        loadProviders(); // Обновить список провайдеров
    });
});
```

### 8. Установка зависимостей и запуск проекта

Сначала установите зависимости, выполнив в терминале команду:

```bash
npm install
```

Затем запустите сервер командой:

```bash
npm start
```

Теперь откройте в браузере `http://localhost:3000`, и вы увидите приложение, где можно добавлять провайдеров, заполняя данные в форму, и просматривать их список на странице.
