<script>
    const board = document.getElementById('game');
    const restartButton = document.getElementById('restart');
    const openProspectButton = document.getElementById('openProspect');
    let currentPlayer = 'X';
    let boardState = Array(9).fill(null);

    board.addEventListener('click', (event) => {
        const index = event.target.dataset.index;
        if (!index || boardState[index]) return;

        boardState[index] = currentPlayer;
        event.target.textContent = currentPlayer;
        currentPlayer = currentPlayer === 'X' ? 'O' : 'X';
        checkWinner();
    });

    restartButton.addEventListener('click', () => {
        boardState.fill(null);
        [...board.children].forEach(cell => cell.textContent = '');
        currentPlayer = 'X';
    });

    openProspectButton.addEventListener('click', () => {
        window.open('prospect.html', '_blank');
    });

    function checkWinner() {
        const winPatterns = [
            [0, 1, 2], [3, 4, 5], [6, 7, 8], // rows
            [0, 3, 6], [1, 4, 7], [2, 5, 8], // columns
            [0, 4, 8], [2, 4, 6]  // diagonals
        ];

        for (let pattern of winPatterns) {
            const [a, b, c] = pattern;
            if (boardState[a] && boardState[a] === boardState[b] && boardState[a] === boardState[c]) {
                alert(`${boardState[a]} победил!`);
                return;
            }
        }

        if (boardState.every(cell => cell)) {
            alert('Ничья!');
        }
    }
</script>
