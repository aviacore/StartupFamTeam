    <script>
        const gameGrid = document.getElementById('gameGrid');
        const resetButton = document.getElementById('resetButton');
        const openProspectButton = document.getElementById('openProspectButton');
        const gridSize = 10;
        let gameData = { hits: [], misses: [], cells: [] }; // Сбрасываемые данные игры

        // Инициализация игры
        function initializeGame() {
            gameGrid.innerHTML = ''; // Очищаем сетку перед новой игрой
            gameData.hits = [];
            gameData.misses = [];
            gameData.cells = [];

            for (let i = 0; i < gridSize * gridSize; i++) {
                const cell = document.createElement('div');
                cell.classList.add('game-cell');
                cell.dataset.index = i; // Добавляем индекс для идентификации
                cell.addEventListener('click', () => handleCellClick(i));
                gameGrid.appendChild(cell);
                gameData.cells.push(cell); // Сохраняем ссылку на ячейку
            }
        }

        // Обработка клика по ячейке
        function handleCellClick(index) {
            // Проверяем, не была ли ячейка уже атакована
            if (!gameData.hits.includes(index) && !gameData.misses.includes(index)) {
                const cell = gameData.cells[index];
                // Симуляция попадания или промаха (можно заменить на реальную логику)
                const isHit = Math.random() > 0.55; // Немного увеличим шанс промаха для интереса

                if (isHit) {
                    cell.classList.add('hit');
                    gameData.hits.push(index);
                    // В реальной игре здесь была бы проверка на уничтожение корабля
                } else {
                    cell.classList.add('miss');
                    gameData.misses.push(index);
                }

                // Проверка на конец игры (например, если все ячейки атакованы)
                if (gameData.hits.length + gameData.misses.length === gridSize * gridSize) {
                    setTimeout(() => alert("Игра окончена! Все ячейки атакованы."), 1000);
                }
            }
        }

        // Сброс игры
        function resetGame() {
            // Удаляем классы hit/miss со всех ячеек
