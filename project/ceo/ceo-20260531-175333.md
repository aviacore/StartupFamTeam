    <button id="restart">Начать заново</button>
    <button id="learnMode">Учить алгоритм</button>
    <script>
        const boardElement = document.getElementById('board');
        const cells = document.querySelectorAll('.cell');
        const restartButton = document.getElementById('restart');
        const learnModeButton = document.getElementById('learnMode');
        let board = Array(9).fill(null);
        let currentPlayer = 'X';
        let learnMode = false;
        let history = [];

        cells.forEach(cell => cell.addEventListener('click', handleCellClick));
        restartButton.addEventListener('click', resetBoard);
        learnModeButton.addEventListener('click', toggleLearnMode);

        function handleCellClick(e) {
            const index = e.target.dataset.index;
            if (board[index] !== null) return;
            board[index] = currentPlayer;
            e.target.textContent = currentPlayer;
            history.push([...board]);
            if (checkWin()) {
                setTimeout(() => alert(`Выиграл игрок ${currentPlayer}`), 10);
                return;
            }
            currentPlayer = currentPlayer === 'X' ? 'O' : 'X';
            if (learnMode) makeSmartMove();
        }

        function checkWin() {
            const winPatterns = [
                [0, 1, 2], [3, 4, 5], [6, 7, 8], // rows
                [0, 3, 6], [1, 4, 7], [2, 5, 8], // columns
                [0, 4, 8], [2, 4, 6]             // diagonals
            ];
            return winPatterns.some(pattern => 
                board[pattern[0]] === currentPlayer &&
                board[pattern[0]] === board[pattern[1]] &&
                board[pattern[0]] === board[pattern[2]]
            );
        }

        function resetBoard() {
            board.fill(null);
            cells.forEach(cell => cell.textContent = '');
            currentPlayer = 'X';
            history = [];
        }

        function toggleLearnMode() {
            learnMode = !learnMode;
            alert(`Режим обучения: ${learnMode ? 'включен' : 'выключен'}`);
        }

        function makeSmartMove() {
            const emptyCells = board.map((val, index) => val === null ? index : null).filter(val => val !== null);
            if (emptyCells.length === 0) return;

            let bestMove = emptyCells[0];
            let bestScore = -Infinity;

            emptyCells.forEach(index => {
                board[index] = currentPlayer;
                const score = minimax(false);
                board[index] = null;
                if (score > bestScore) {
                    bestScore = score;
                    bestMove = index;
                }
            });

            board[bestMove] = currentPlayer;
            cells[bestMove].textContent = currentPlayer;
            if (checkWin()) {
                setTimeout(() => alert(`Выиграл игрок ${currentPlayer}`), 10);
                return;
            }
            currentPlayer = currentPlayer === 'X' ? 'O' : 'X';
        }

        function minimax(isMaximizing) {
            if (checkWin()) return isMaximizing ? -1 : 1;

            const emptyCells = board.map((val, index) => val === null ? index : null).filter(val => val !== null);
            if (emptyCells.length === 0) return 0;

            let bestScore = isMaximizing ? -Infinity : Infinity;

            emptyCells.forEach(index => {
                board[index] = isMaximizing ? currentPlayer : (currentPlayer === 'X' ? 'O' : 'X');
                const score = minimax(!isMaximizing);
                board[index] = null;
                if (isMaximizing) {
                    bestScore = Math.max(score, bestScore);
                } else {
                    bestScore = Math.min(score, bestScore);
                }
            });

            return bestScore;
        }
