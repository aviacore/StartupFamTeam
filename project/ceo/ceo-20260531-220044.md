        const playerGrid = document.getElementById('player-grid');

        const GRID_SIZE = 10;

        let playerBoardState = Array(GRID_SIZE).fill(0).map(() => Array(GRID_SIZE).fill(0));
        // 0: empty, 2: hit, 3: miss (opponent board only tracks player's shots)
        let opponentBoardState = Array(GRID_SIZE).fill(0).map(() => Array(GRID_SIZE).fill(0));

        let playerShips = []; // Stores {id, size, cells: [{row, col}], orientation, hits: 0, sunk: false}
        let opponentShips = []; // Stores actual opponent ship placements (for simulation)
        let currentTurn = 'player';
        let gameOver = false;

        // Corresponds to AGENTS.md roles
        const shipConfigs = [
             { size: 4, count: 1, id: 'ceo' },
             { size: 3, count: 2, id: 'pm_cto' },

        let allShipSizes = []; // Flat list of sizes to place, e.g., [4, 3, 3, 2, 2, 2, 1, 1, 1, 1]

        function initializeShipConfigs() {
            shipConfigs.forEach(config => {
                for (let i = 0; i < config.count; i++) {
                    allShipSizes.push(config.size);
                 }
                totalShipsToPlace += config.count;
            });
            allShipSizes.sort((a, b) => b - a); // Place larger ships first
        }

        function initializeGrid(gridElement, stateArray, isPlayerGrid = false) {
            gridElement.innerHTML = '';
            for (let r = 0; r < GRID_SIZE; r++) {
                for (let c = 0; c < GRID_SIZE; c++) {
                    const cell = document.createElement('div');
                    cell.classList.add('grid-cell');
                    cell.dataset.row = r;
                    cell.dataset.col = c;

                    if (isPlayerGrid) {
                        cell.addEventListener('click', handlePlayerCellClick);
                    } else {
                        cell.addEventListener('click', handleOpponentCellClick);
                    }

                    updateCellAppearance(cell, stateArray[r][c], isPlayerGrid);
                    gridElement.appendChild(cell);
                }
            }
        }

        function updateCellAppearance(cell, stateType, isPlayerGrid) {
             cell.classList.remove('ship', 'hit', 'miss', 'sunk', 'temp-ship');
             cell.style.backgroundImage = '';
             cell.style.backgroundColor = '';

             const row = parseInt(cell.dataset.row);
             const col = parseInt(cell.dataset.col);

             switch(stateType) {
                 case 0: // Empty
                     break;
                 case 1: // Ship (player board)
                     cell.classList.add('ship');
                     break;
                 case 2: // Hit
                     cell.classList.add('hit');
                     break;
                 case 3: // Miss
                     cell.classList.add('miss');
                     break;
                 case 5: // Temporary ship (player board only)
                    if (isPlayerGrid) cell.classList.add('temp-ship');
                    break;
             }
        }

        function getCellElement(row, col, gridElement) {
            return gridElement.querySelector(`[data-row="${row}"][data-col="${col}"]`);
        }

        function isValidCoordinate(row, col) {
            return row >= 0 && row < GRID_SIZE && col >= 0 && col < GRID_SIZE;
        }

        function canPlaceShip(row, col, size, orientation, boardState) {
            if (orientation === 'horizontal') {
                if (col + size > GRID_SIZE) return false;
            } else {
                if (row + size > GRID_SIZE) return false;
            }

            for (let i = 0; i < size; i++) {
                let r = row, c = col;
                if (orientation === 'horizontal') c += i; else r += i;

                if (boardState[r][c] === 1) return false;

                for (let dr = -1; dr <= 1; dr++) {
                    for (let dc = -1; dc <= 1; dc++) {
                        const nr = r + dr;
                        const nc = c + dc;

                        if (isValidCoordinate(nr, nc)) {
                            let isPartOfCurrentShip = false;
                            for (let k = 0; k < size; k++) {
                                let sr = row, sc = col;
                                if (orientation === 'horizontal') sc += k; else sr += k;
                                if (nr === sr && nc === sc) {
                                    isPartOfCurrentShip = true;
                                    break;
                                }
                            }
                            if (!isPartOfCurrentShip && boardState[nr][nc] === 1) {
                                return false;
                            }
                        }
                    }
                }
            }
            return true;
        }

        function placeShip(shipData, boardState, shipsArray) {
            shipData.cells.forEach(cellPos => {
                boardState[cellPos.row][cellPos.col] = 1;
                const gridCell = getCellElement(cellPos.row, cellPos.col, playerGrid);
                if (gridCell) {
                     gridCell.classList.remove('temp-ship');
                     gridCell.classList.add('ship');
                     gridCell.onclick = null;
                }
            });
            shipsArray.push(shipData);
            placedShipCount++;
         }

        function drawTemporaryShipPreview(shipCells) {
            temporaryShipPlacement.forEach(({row, col}) => {
                 const cell = getCellElement(row, col, playerGrid);
                 if (cell) {
                    cell.classList.remove('temp-ship');
                    if(playerBoardState[row][col] !== 1 && playerBoardState[row][col] !== 2 && playerBoardState[row][col] !== 3) {
                        playerBoardState[row][col] = 0;
                        updateCellAppearance(cell, 0, true);
                    }
                 }
            });
            temporaryShipPlacement = [];

            shipCells.forEach(cellPos => {
                const cell = getCellElement(cellPos.row, cellPos.col, playerGrid);
                if (cell && isValidCoordinate(cellPos.row, cellPos.col)) {
                    playerBoardState[cellPos.row][cellPos.col] = 5;
                    cell.classList.add('temp-ship');
                    temporaryShipPlacement.push({row: cellPos.row, col: cellPos.col});
                }
            });
        }
