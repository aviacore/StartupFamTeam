## 1. Критический обзор кода Developer: найденные баги и уязвимости

Я провёл статический анализ кода из `project/src/`. Вот что я обнаружил:

### Баг #1: Утечка памяти в `roll-call-manager.js` (использование `setTimeout` без очистки)

**Описание:** Функция `startRollCall` создаёт `setTimeout` на 10 секунд. Если CEO нажмёт кнопку «Перекличка» несколько раз подряд (до истечения таймаута), будет создано несколько параллельных таймеров. Они все выполнятся, каждый попытается закрыть перекличку и вставить записи в БД.

**Риски:**
- Удвоение/утроение записей в `roll_call_responses`.
- Некорректный статус переклички.
- Утечка памяти (таймеры не очищаются).

**Фикс:**
```javascript
// roll-call-manager.js
let activeTimeouts = new Map(); // { rollCallId: timeoutId }

function startRollCall(wss, connectedAgents, rollCallId, db) {
  // Очищаем предыдущий таймер для этого ID, если был
  if (activeTimeouts.has(rollCallId)) {
    clearTimeout(activeTimeouts.get(rollCallId));
  }

  const timeoutId = setTimeout(() => {
    // ... логика закрытия ...
    activeTimeouts.delete(rollCallId); // очищаем после выполнения
  }, TIMEOUT_MS);

  activeTimeouts.set(rollCallId, timeoutId);
}
```

### Баг #2: Отсутствие валидации на сервере (`server.js`)

**Описание:** Сервер принимает любые JSON-сообщения от WebSocket-клиентов без проверки `type`. Агент может отправить сообщение с произвольным `rollCallId`, `role`, `status`, и сервер запишет это в БД. Злоумышленник (или сломанный агент) может "подделать" ответ от другого агента.

**Риски:**
- Нарушение целостности данных.
- Ошибки в аналитике (PM будет видеть ложные статусы).
- Невозможность доверять логу.

**Фикс (добавить в `server.js` в обработчик `ws.on('message')`):**
```javascript
ws.on('message', (message) => {
  try {
    const data = JSON.parse(message);
    
    // Валидация обязательных полей
    if (!data.type || !data.rollCallId || !data.role) {
      console.warn('Некорректное сообщение от агента:', data);
      ws.send(JSON.stringify({ type: 'error', message: 'Invalid message format. Required: type, rollCallId, role' }));
      return;
    }

    // Валидация значения status (белый список)
    const allowedStatuses = ['готов', 'не готов', 'в процессе'];
    if (data.type === 'pong' && data.status && !allowedStatuses.includes(data.status)) {
      console.warn('Неизвестный статус:', data.status);
      ws.send(JSON.stringify({ type: 'error', message: `Invalid status: ${data.status}. Allowed: ${allowedStatuses.join(', ')}` }));
      return;
    }

    // Проверка, что агент с таким role зарегистрирован
    const agentInfo = connectedAgents.get(ws);
    if (!agentInfo || agentInfo.role !== data.role) {
      console.warn('Попытка ответить от чужого имени:', data.role);
      ws.send(JSON.stringify({ type: 'error', message: 'Role mismatch. You can only respond as your registered role.' }));
      return;
    }

    // Обработка по типу сообщения
    if (data.type === 'register') {
      connectedAgents.set(ws, { role: data.role, connectedAt: new Date() });
      ws.send(JSON.stringify({ type: 'registered', role: data.role }));
    } else if (data.type === 'pong') {
      processAgentResponse(ws, data, db);
    }
  } catch (e) {
    console.error('Ошибка парсинга JSON:', e);
    ws.send(JSON.stringify({ type: 'error', message: 'Invalid JSON' }));
  }
});
```

### Баг #3: Нет обработки случая, когда БД не инициализирована

**Описание:** В `server.js` функция `initDB()` вызывается в `app.post('/api/roll-call')` и `app.get('/api/status')`. Если сервер только запущен и БД ещё не создана (например, первый запуск), первый запрос может упасть с ошибкой «no such table: roll_calls».

**Риски:** HTTP 500 на первый запрос.

**Фикс:** Вызывать `initDB()` при старте сервера, а не в каждом роуте:
```javascript
server.listen(PORT, () => {
  console.log(`Сервер запущен на порту ${PORT}`);
  const db = initDB(); // Создаём таблицы один раз
  app.locals.db = db;  // Сохраняем в app.locals для доступа из роутов
});
```

### 3.1. `tests/smoke-test.sh` (ручной smoke-тест для CEO)

```bash
#!/bin/bash
# Smoke-тест RollCall AI
echo "=== Smoke Test ==="

# 1. Проверка, что сервер запущен
if curl -s http://localhost:3000 > /dev/null; then
  echo "✅ Сервер доступен на порту 3000"
else
  echo "❌ Сервер не отвечает. Запустите: node server.js"
  exit 1
fi

# 2. Проверка REST API
STATUS=$(curl -s http://localhost:3000/api/status | grep -o '"status":"[^"]*"' | head -1)
if [ -n "$STATUS" ]; then
  echo "✅ API /api/status работает. Статус: $STATUS"
else
  echo "❌ API /api/status недоступен"
  exit 1
fi

# 3. Запуск одного агента (в фоне)
node agents/ideation.js &
AGENT_PID=$!
sleep 1
echo "✅ Агент запущен (PID: $AGENT_PID)"

# 4. Проведение переклички
echo "Провожу перекличку..."
RESPONSE=$(curl -s -X POST http://localhost:3000/api/roll-call)
ROLL_CALL_ID=$(echo $RESPONSE | grep -o '"rollCallId":[0-9]*' | grep -o '[0-9]*')
if [ -n "$ROLL_CALL_ID" ]; then
  echo "✅ Перекличка #$ROLL_CALL_ID создана"
else
  echo "❌ Не удалось создать перекличку"
  kill $AGENT_PID 2>/dev/null
  exit 1
fi

# 5. Ожидание завершения
sleep 12
FINAL_STATUS=$(curl -s http://localhost:3000/api/status | grep -o '"status":"[^"]*"' | head -1)
if echo "$FINAL_STATUS" | grep -q "closed"; then
  echo "✅ Перекличка #$ROLL_CALL_ID завершена (статус: closed)"
else
  echo "❌ Перекличка не завершена. Текущий статус: $FINAL_STATUS"
fi

# Очистка
kill $AGENT_PID 2>/dev/null
echo "=== Smoke Test Completed ==="
```

### 3.2. `tests/load-test.js` (автоматический нагрузочный тест на Node.js)

Я бы рекомендовал использовать `autocannon` или `artillery`, но для MVP подойдёт простой скрипт:

```javascript
// tests/load-test.js
const WebSocket = require('ws');
const https = require('http');

const SERVER_URL = 'ws://localhost:3000';
const NUM_AGENTS = 10;

console.log(`Запуск нагрузочного теста: ${NUM_AGENTS} агентов`);

// Создаём 10 агентов
const agents = [];
for (let i = 0; i < NUM_AGENTS; i++) {
  const ws = new WebSocket(SERVER_URL);
  const role = `Агент-${i + 1}`;
  
  ws.on('open', () => {
    ws.send(JSON.stringify({ type: 'register', role }));
  });
  
  ws.on('message', (message) => {
    const data = JSON.parse(message);
    if (data.type === 'ping') {
      // Имитируем ответ через 200-300 мс
      setTimeout(() => {
        ws.send(JSON.stringify({
          type: 'pong',
          rollCallId: data.rollCallId,
          role,
          status: 'готов',
          message: 'Нагрузочный тест'
        }));
      }, 200 + Math.random() * 100);
    }
  });
  
  agents.push(ws);
}

// Запускаем 5 перекличек подряд
let completedCalls = 0;
const runRollCall = () => {
  https.get('http://localhost:3000/api/roll-call', (resp) => {
    let data = '';
    resp.on('data', (chunk) => data += chunk);
    resp.on('end', () => {
      const { rollCallId } = JSON.parse(data);
      console.log(`Перекличка #${rollCallId} запущена`);
      
      setTimeout(() => {
        completedCalls++;
        console.log(`Перекличка #${rollCallId} завершена (всего: ${completedCalls})`);
        
        if (completedCalls < 5) {
          runRollCall();
        } else {
          console.log('Нагрузочный тест завершён');
          agents.forEach(ws => ws.close());
          process.exit(0);
        }
      }, 12000);
    });
  });
};

setTimeout(runRollCall, 1000);
```
