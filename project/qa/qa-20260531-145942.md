            if (!this.memory.has(key)) return null;
            const record = this.memory.get(key);
            return record.move;
        }

        remember(board, move) {
            const key = this.boardKey(board);
            if (!this.memory.has(key)) {
                this.memory.set(key, { move: move, wins: 0, losses: 0, draws: 0, seen: 0 });
            }
            const rec = this.memory.get(key);
            rec.seen = (rec.seen || 0) + 1;
            if (board[move] !== null) return;
            if (rec.move !== move && rec.wins < 1) {
                rec.move = move;
            }
        }

        learnFromGame(boardHistory, winner) {
            this.gamesPlayed++;
            if (winner === 'O') this.wins++;
            else if (winner === 'draw') this.draws++;
            else if (winner === 'X') this.losses++;

            for (let i = 0; i < boardHistory.length; i++) {
                const state = boardHistory[i];
                const key = this.boardKey(state.board);
                if (this.memory.has(key)) {
                    const rec = this.memory.get(key);
                    if (winner === 'O') {
                        rec.wins = (rec.wins || 0) + 1;
                    } else if (winner === 'X') {
                        rec.losses = (rec.losses || 0) + 1;
                    } else {
                        rec.draws = (rec.draws || 0) + 1;
                    }
                }
            }

    *   **Серьезность:** Высокая (делает игру неиграбельной в некоторых сценариях).

    *   **Шаги воспроизведения:**
