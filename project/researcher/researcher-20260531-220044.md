        function initializeBoard(gridElement, boardArray) {
            gridElement.innerHTML = '';
            for (let r = 0; r < GRID_SIZE; r++) {
                for (let c = 0; c < GRID_SIZE; c++) {
                    const cell = document.createElement('div');
                    cell.classList.add('grid-cell');
                    cell.dataset.row = r;
                    cell.dataset.col = c;
                    gridElement.appendChild(cell);
                    if (boardArray) {
                        boardArray[r] = boardArray[r] || [];
                        boardArray[r][c] = 0; // 0: empty, 1: ship, 2: hit, 3: miss
                    }
                }
            }
        }

        function createShip(size, row, col, orientation) {
            const ship = {
                size: size,
                hits: 0,
                cells: [],
                orientation: orientation
            };
            for (let i = 0; i < size; i++) {
                let r = row, c = col;
                if (orientation === 'horizontal') {
                    c += i;
                } else {
                    r += i;
                }
                ship.cells.push({ row: r, col: c });
                playerBoard[r][c] = 1; // Mark as ship on player board
            }
            return ship;
        }

        function canPlaceShip(size, row, col, orientation) {
            for (let i = 0; i < size; i++) {
                let r = row, c = col;
                if (orientation === 'horizontal') {
                    c += i;
                } else {
                    r += i;
                }

                if (r < 0 || r >= GRID_SIZE || c < 0 || c >= GRID_SIZE) return false; // Out of bounds

                // Check surrounding cells
                for (let dr = -1; dr <= 1; dr++) {
                    for (let dc = -1; dc <= 1; dc++) {
                        const nr = r + dr;
                        const nc = c + dc;
                        if (nr >= 0 && nr < GRID_SIZE && nc >= 0 && nc < GRID_SIZE) {
                            if (playerBoard[nr][nc] === 1) {
                                // Check if the overlapping cell is part of the *current* potential placement
                                let isPartofCurrent = false;
                                for(let j=0; j<i; j++){
                                     let cr = row, cc = col;
                                     if (orientation === 'horizontal') cc += j; else cr += j;
                                     if(nr === cr && nc === cc) isPartofCurrent = true;
                                }
                                if (!isPartofCurrent) return false; // Collision with another ship
                            }
                        }
                    }
                }
            }
            return true;
        }

            placedShips.push(ship);
             ship.cells.forEach(cell => {
                 playerBoard[cell.row][cell.col] = 1;
                 const gridCell = playerGrid.querySelector(`[data-row="${cell.row}"][data-col="${cell.col}"]`);
                 if (gridCell) {
                     gridCell.classList.add('ship');
                 }
             });
        }

                 if (gridCell) {
                    gridCell.classList.remove('ship', 'temp-ship');
                 }
             });
             tempShipPlacement = [];
        }

                     if (gridCell) {
                         gridCell.classList.add('ship', 'temp-ship');
                     }
                     tempShipPlacement.push({row, col});
                 }
             });
        }

                 if (isBoardFullOfHits(opponentBoard)) { // Simplified win condition

                 while (!success) {
