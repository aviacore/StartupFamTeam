    <div class="game-container">
        <h1>Крестики-Нолики</h1>

            <button class="restart-button" onclick="restartGame()">Начать заново</button>
            <div class="message" id="message">Ход Игрока (X)</div>
            <div class="score-board">
                Игрок (X): <span id="player-score">0</span> | Компьютер (O): <span id="ai-score">0</span>
            </div>

    <script>
        const grid = document.getElementById('game-grid');
        const messageElement = document.getElementById('message');
        const playerScoreElement = document.getElementById('player-score');
        const aiScoreElement = document.getElementById('ai-score');
        let board = Array(9).fill(null);
        let currentPlayer = 'X';
        let gameActive = true;
        let playerScore = 0;
        let aiScore = 0;

        const winningConditions = [
            [0, 1, 2], [3, 4, 5], [6, 7, 8], // Rows
            [0, 3, 6], [1, 4, 7], [2, 5, 8], // Columns
            [0, 4, 8], [2, 4, 6]             // Diagonals
        ];

        function handleCellClick(event) {
            const index = parseInt(event.target.dataset.index);

            if (board[index] === null && gameActive) {
                board[index] = currentPlayer;
                event.target.classList.add(currentPlayer);
                event.target.textContent = currentPlayer;

                if (checkWin(currentPlayer)) {
                    if (currentPlayer === 'X') {
                        playerScore++;
                        playerScoreElement.textContent = playerScore;
                        messageElement.textContent = 'Игрок победил!';
                    } else {
                        aiScore++;
                        aiScoreElement.textContent = aiScore;
                        messageElement.textContent = 'Компьютер победил!';
                    }
                    gameActive = false;
                } else if (board.every(cell => cell !== null)) {
                    messageElement.textContent = 'Ничья!';
                    gameActive = false;
                } else {
                    currentPlayer = currentPlayer === 'X' ? 'O' : 'X';
                    messageElement.textContent = `Ход Игрока (${currentPlayer})`;
                    if (currentPlayer === 'O' && gameActive) {
                        setTimeout(() => aiMove(), 700); // Небольшая задержка для хода ИИ
                    }
                }
            }
        }

        function checkWin(player) {
            for (const condition of winningConditions) {
                const [a, b, c] = condition;
                if (board[a] === player && board[b] === player && board[c] === player) {
                    return true;
                }
            }
            return false;
        }

        function restartGame() {
            board.fill(null);
            currentPlayer = 'X';
            gameActive = true;
            messageElement.textContent = `Ход Игрока (${currentPlayer})`;
            document.querySelectorAll('.cell').forEach(cell => {
                cell.classList.remove('X', 'O');
                cell.textContent = '';
            });
        }

        function aiMove() {
            // Простая "самообучающаяся" логика:
            // 1. Пытается выиграть
            // 2. Пытается заблокировать игрока
            // 3. Выбирает случайную свободную клетку (с приоритетом центра, затем углов)

            let bestMove = -1;

            // 1. Попытка выиграть
            for (let i = 0; i < board.length; i++) {
                if (board[i] === null) {
                    const tempBoard = [...board];
                    tempBoard[i] = 'O';
                    if (checkWinOnBoard(tempBoard, 'O')) {
                        bestMove = i;
                        break;
                    }
                }
            }

            // 2. Попытка заблокировать игрока
            if (bestMove === -1) {
                for (let i = 0; i < board.length; i++) {
                    if (board[i] === null) {
                        const tempBoard = [...board];
                        tempBoard[i] = 'X'; // Предполагаем ход игрока
                        if (checkWinOnBoard(tempBoard, 'X')) {
                            bestMove = i;
                            break;
                        }
                    }
                }
            }

            // 3. Выбор лучшей доступной клетки
            if (bestMove === -1) {
                const preferredMoves = [4, 0, 2, 6, 8, 1, 3, 5, 7]; // Центр, углы, стороны
                for (const move of preferredMoves) {
                    if (board[move] === null) {
                        bestMove = move;
                        break;
                    }
                }
            }

            if (bestMove !== -1) {
                board[bestMove] = 'O';
                const cell = grid.children[bestMove];
                cell.classList.add('O');
                cell.textContent = 'O';

                if (checkWin('O')) {
                    aiScore++;
                    aiScoreElement.textContent = aiScore;
                    messageElement.textContent = 'Компьютер победил!';
                    gameActive = false;
                } else if (board.every(cell => cell !== null)) {
                    messageElement.textContent = 'Ничья!';
                    gameActive = false;
                } else {
                    currentPlayer = 'X';
                    messageElement.textContent = `Ход Игрока (${currentPlayer})`;
                }
            }
        }

        function checkWinOnBoard(currentBoard, player) {
            for (const condition of winningConditions) {
                const [a, b, c] = condition;
                if (currentBoard[a] === player && currentBoard[b] === player && currentBoard[c] === player) {
                    return true;
                }
            }
            return false;
        }


        document.querySelectorAll('.cell').forEach(cell => {
            cell.addEventListener('click', handleCellClick);
        });

        restartGame(); // Инициализация игры при загрузке
    </script>
