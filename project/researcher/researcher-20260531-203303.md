        <div class="grid" id="grid"></div>

            <button id="resetButton">Новая игра</button>
        </div>
    </div>
    <script>
        const grid = document.getElementById('grid');
        const resetButton = document.getElementById('resetButton');
        const gridSize = 10;
        let hits = [];
        let misses = [];
        
        function createGrid() {
            for (let i = 0; i < gridSize * gridSize; i++) {
                const cell = document.createElement('div');
                cell.classList.add('cell');
                cell.addEventListener('click', () => makeGuess(i));
                grid.appendChild(cell);
            }
        }

        function makeGuess(index) {
            if (!hits.includes(index) && !misses.includes(index)) {
                // Simulate a hit or miss for demonstration
                const isHit = Math.random() > 0.5;
                if (isHit) {
                    hits.push(index);
                    grid.children[index].classList.add('hit');
                } else {
                    misses.push(index);
                    grid.children[index].classList.add('miss');
                }
            }
        }

        function resetGame() {
            hits = [];
            misses = [];
            Array.from(grid.children).forEach(cell => {
                cell.classList.remove('hit', 'miss');
            });
        }

        resetButton.addEventListener('click', resetGame);
        createGrid();
    </script>
