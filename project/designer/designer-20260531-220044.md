        function getCellElement(row, col, gridElement) {
            return gridElement.querySelector(`[data-row="${row}"][data-col="${col}"]`);
        }

        function isValidCoordinate(row, col) {
            return row >= 0 && row < GRID_SIZE && col >= 0 && col < GRID_SIZE;
        }

        function checkSurroundingCells(row, col, size, orientation, boardState) {
            for (let i = 0; i < size; i++) {
                let r = row, c = col;
                if (orientation === 'horizontal') c += i;
                else r += i;

                for (let dr = -1; dr <= 1; dr++) {
                    for (let dc = -1; dc <= 1; dc++) {
                        const nr = r + dr;
                        const nc = c + dc;

                        if (isValidCoordinate(nr, nc)) {
                            if (boardState[nr][nc] === 1) { // If there's a ship part
                                // Check if this neighboring ship part belongs to the *current* ship being placed
                                let isPartOfCurrentShip = false;
                                for (let j = 0; j < size; j++) {
                                    let cr = row, cc = col;
                                    if (orientation === 'horizontal') cc += j; else cr += j;
                                    if (nr === cr && nc === cc) {
                                        isPartOfCurrentShip = true;
                                        break;
                                    }
                                }
                                if (!isPartOfCurrentShip) return false; // Collision with another ship
                            }
                        }
                    }
                }
            }
            return true;
        }


         function canPlaceShip(row, col, size, orientation, boardState) {
            // Check bounds
            if (orientation === 'horizontal') {
                if (col + size > GRID_SIZE) return false;
            } else {
                if (row + size > GRID_SIZE) return false;
            }

            // Check for collisions with existing ships and ensure valid spacing
            for (let i = 0; i < size; i++) {
                let r = row, c = col;
                if (orientation === 'horizontal') c += i; else r += i;

                if (boardState[r][c] === 1) return false; // Direct overlap

                // Check surrounding cells for adjacent ships (standard rule)
                for (let dr = -1; dr <= 1; dr++) {
                    for (let dc = -1; dc <= 1; dc++) {
                        const nr = r + dr;
                        const nc = c + dc;

                        if (isValidCoordinate(nr, nc)) {
                             // Ensure we are not checking our own cells if they are part of the current placement
                            let isPartOfCurrentPlacement = false;
                            for(let k=0; k<size; k++) {
                                let sr = row, sc = col;
                                if(orientation === 'horizontal') sc += k; else sr += k;
                                if(nr === sr && nc === sc) {
                                    isPartOfCurrentPlacement = true;
                                    break;
                                }
                            }
                            if (!isPartOfCurrentPlacement && boardState[nr][nc] === 1) {
                                return false; // Adjacent to another ship
                            }
                        }
                    }
                }
            }
            return true;
        }

        function placeShip(shipData, boardState, shipsArray) {
            shipData.cells.forEach(cellPos => {
                boardState[cellPos.row][cellPos.col] = 1; // Mark as ship
                const gridCell = getCellElement(cellPos.row, cellPos.col, playerGrid);
                if (gridCell) {
                     gridCell.classList.add('ship');
                     gridCell.style.backgroundColor = ''; // Reset any temp color
                     gridCell.onclick = null; // Disable clicking on placed ship cells
                }
            });
            shipsArray.push(shipData);
         }

        function drawTemporaryShipPreview(ship) {
            // Clear previous temporary placement visuals
             temporaryShipPlacement.forEach(({row, col}) => {
                 const cell = getCellElement(row, col, playerGrid);
                 if (cell) {
                    cell.classList.remove('temp-ship');
                    if (playerBoardState[row][col] !== 1 && playerBoardState[row][col] !== 2 && playerBoardState[row][col] !== 3){ // Only reset if not hit/miss/ship
                        if(boardState[row][col] === 0) cell.style.backgroundColor = ''; // Reset to default if empty
                    }
                 }
             });
             temporaryShipPlacement = [];

            // Draw new temporary placement
            ship.cells.forEach(cellPos => {
                const cell = getCellElement(cellPos.row, cellPos.col, playerGrid);
                if (cell && isValidCoordinate(cellPos.row, cellPos.col)) {
                    playerBoardState[cellPos.row][cellPos.col] = 5; // Mark as temporary
                    cell.classList.add('temp-ship');
                    cell.style.backgroundColor = ''; // Ensure default bg is not overriding class
                    temporaryShipPlacement.push({row: cellPos.row, col: cellPos.col});
                }
            });
        }

        function updateStatusMessage() {
            const currentShipSize = ALL_SHIP_SIZES[currentShipToPlaceIndex];
            if (!currentShipSize) {
                statusMessage.textContent = `Все корабли (${placedShipCount}/${totalShipsToPlace}) расставлены!`;
                return;
            }
            statusMessage.textContent = `Размещение: ${currentShipSize}-клеточный корабль. Ориентация: ${currentShipOrientation.charAt(0).toUpperCase() + currentShipOrientation.slice(1)}`;
        }
