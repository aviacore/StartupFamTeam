## Вклад UI/UX Designer: Превращаем функциональность в эстетику

### 1. Дизайн-система: основа визуального единства

Вместо разрозненных стилей, я предлагаю создать единую дизайн-систему, которая будет использоваться во всех компонентах: от веб-интерфейса до CLI-уведомлений.

**Основные принципы:**
- **Тёмная тема** — современный стандарт, снижает нагрузку на глаза, подходит для работы в любое время суток.
- **Минимализм** — только нужные элементы, никакого визуального шума.
- **Плавность** — каждая анимация должна быть функциональной и естественной.
- **Адаптивность** — интерфейс должен одинаково хорошо выглядеть на десктопе, планшете и мобильном телефоне.

**Цветовая палитра:**
- **Фон:** `#0F0F1A` (глубокий тёмно-синий, почти чёрный) — создаёт ощущение бесконечности и технологичности.
- **Основной акцент:** `#6C63FF` (фиолетовый) — символизирует креативность, инновации и командную работу.
- **Второй акцент:** `#00D9FF` (голубой) — добавляет динамику и контраст.
- **Текст основной:** `#E0E0E0` (светло-серый) — мягкий, не утомляет глаза.
- **Текст вторичный:** `#888888` (серый) — для подсказок и дат.
- **Успех:** `#4CAF50` (зелёный).
- **Ошибка:** `#FF5252` (красный).

**Типографика:**
- **Основной шрифт:** `'Inter', sans-serif` — современный, читаемый, отлично подходит для UI.
- **Моноширинный шрифт для кода:** `'Fira Code', monospace` — для отображения JSON и логов.
- **Размеры:** `14px` (основной текст), `16px` (акцентный), `20px` (заголовки), `32px` (основной заголовок).

**Тени и эффекты:**
- **Тень для карточек:** `0 8px 32px rgba(0, 0, 0, 0.4)` — создаёт глубину.
- **Свечение для акцентов:** `0 0 20px rgba(108, 99, 255, 0.5)` — привлекает внимание к важным элементам.

    // --- Функция обновления статуса с анимацией ---
    function updateStatus(message, type = 'info') {
      const statusEl = document.getElementById('status');
      statusEl.className = type;
      statusEl.innerHTML = `<p>${message}</p>`;
      statusEl.style.animation = 'none';
      statusEl.offsetHeight; // reflow
      statusEl.style.animation = 'slideIn 0.3s ease';
    }

    async function loadResponses() {
      try {
        const res = await fetch('/api/agents/responses');
        if (!res.ok) throw new Error('Ошибка загрузки');
        const data = await res.json();
        
        const tbody = document.getElementById('responses-body');
        tbody.innerHTML = '';
        
        if (data.length === 0) {
          tbody.innerHTML = '<tr class="empty-row"><td colspan="4">📭 Пока нет данных. Используйте CLI или API.</td></tr>';
          updateStatus('💡 Данные отсутствуют. Добавьте ответы агентов.', 'info');
          return;
        }
        
        data.forEach((item, index) => {
          const row = document.createElement('tr');
          row.style.animationDelay = `${index * 0.1}s`;
          row.innerHTML = `
            <td class="agent-name">${item.agent}</td>
            <td>${item.russian_title}</td>
            <td class="synonym">${item.synonym || '—'}</td>
            <td>${item.timestamp ? new Date(item.timestamp).toLocaleDateString('ru-RU') : '—'}</td>
          `;
          tbody.appendChild(row);
        });
        
        updateStatus(`✅ Загружено ответов: ${data.length}`, 'success');
      } catch (err) {
        console.error('Ошибка загрузки:', err);
        updateStatus('❌ Не удалось загрузить данные. Проверьте сервер.', 'error');
      }
    }

    async function generatePlan() {
      updateStatus('⏳ Генерация Plan_06.md...', 'info');
      try {
        const res = await fetch('/api/agents/plan06');
        const result = await res.json();
        if (res.ok) {
          updateStatus(`✅ Plan_06.md сгенерирован: ${result.path}`, 'success');
        } else {
          updateStatus(`❌ ${result.error}`, 'error');
        }
      } catch (err) {
        updateStatus(`❌ Ошибка: ${err.message}`, 'error');
      }
    }
